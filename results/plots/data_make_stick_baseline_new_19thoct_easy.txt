    """
    Simple planner that moves the agent to the nearest stick primitive,
    orients it towards the stick and issues a USE action.
    Returns a list of action indices (DOWN=0, UP=1, LEFT=2, RIGHT=3, USE=4).
    """
    # ----- constants -------------------------------------------------
    DOWN, UP, LEFT, RIGHT, USE = 0, 1, 2, 3, 4
    opposite_dir = {DOWN: UP, UP: DOWN, LEFT: RIGHT, RIGHT: LEFT}

    # ----- access current state ---------------------------------------
    state = env._current_state            # CraftState instance
    grid = state.grid                      # (W, H, n_kinds) occupancy array
    pos = list(state.pos)                  # [x, y]
    cur_dir = state.dir                    # current facing direction

    # ----- identify stick primitive ------------------------------------
    stick_name = "stick"
    try:
        stick_idx = env.world.cookbook.index[stick_name]
    except Exception:                     # fallback – cannot find index
        return []

    # locate any cell that contains a stick
    stick_cells = list(zip(*np.where(grid[:, :, stick_idx] > 0)))
    if not stick_cells:
        return []                          # no sticks present

    # pick the first stick (could be improved to nearest)
    tx, ty = stick_cells[0]

    actions = []

    # ----- move until we are orthogonally adjacent to the stick -------
    while abs(pos[0] - tx) + abs(pos[1] - ty) != 1:
        if pos[0] < tx:          # need to move right
            act = RIGHT
            pos[0] += 1
        elif pos[0] > tx:        # need to move left
            act = LEFT
            pos[0] -= 1
        elif pos[1] < ty:        # need to move down
            act = DOWN
            pos[1] += 1
        else:                    # need to move up
            act = UP
            pos[1] -= 1
        cur_dir = act            # moving also sets facing direction
        actions.append(act)

    # ----- face the stick ---------------------------------------------
    dx, dy = tx - pos[0], ty - pos[1]
    if dx == 1:
        need_dir = RIGHT
    elif dx == -1:
        need_dir = LEFT
    elif dy == 1:
        need_dir = DOWN
    else:
        need_dir = UP

    if cur_dir != need_dir:
        # turn by moving opposite then back (no explicit turn action)
        opp = opposite_dir[need_dir]
        actions.append(opp)
        # update temporary position (ignore bounds – world is toroidal enough)
        if opp == DOWN:
            pos[1] += 1
        elif opp == UP:
            pos[1] -= 1
        elif opp == LEFT:
            pos[0] -= 1
        else:  # RIGHT
            pos[0] += 1
        cur_dir = opp

        actions.append(need_dir)
        if need_dir == DOWN:
            pos[1] += 1
        elif need_dir == UP:
            pos[1] -= 1
        elif need_dir == LEFT:
            pos[0] -= 1
        else:  # RIGHT
            pos[0] += 1
        cur_dir = need_dir

    # ----- collect the stick -------------------------------------------
    actions.append(USE)

    return actions

    """
    Simple planner that searches for a short sequence of actions that yields a
    ``stick`` in the agent's inventory.  It performs a breadth‑first search on
    the underlying ``CraftState`` objects using the environment's own transition
    function (`CraftState.step`).  The search depth is limited to keep the
    computation tractable.

    Parameters
    ----------
    env : CraftLab
        The environment instance (already created by the factory).  The function
        will reset the environment, inspect its world to obtain the index of the
        ``stick`` item and then return a list of action integers that lead to a
        state where the inventory contains at least one stick.

    Returns
    -------
    list[int]
        A list of actions (0=DOWN, 1=UP, 2=LEFT, 3=RIGHT, 4=USE) that achieves the
        goal.  If no plan is found within the depth limit, an empty list is
        returned.
    """
    # --------------------------------------------------------------------- #
    # 1. Initialise the environment and obtain the starting CraftState.
    # --------------------------------------------------------------------- #
    env.reset()                     # initialise internal state
    try:
        start_state = env._current_state   # CraftState created by reset()
    except AttributeError:          # fallback – some wrappers may expose it differently
        start_state = getattr(env, "current_state", None)
    if start_state is None:
        raise RuntimeError("Unable to access the current CraftState from the environment.")

    # --------------------------------------------------------------------- #
    # 2. Resolve the integer index that corresponds to the item ``stick``.
    # --------------------------------------------------------------------- #
    # The ``cookbook`` lives inside the world, which is reachable via the env.
    try:
        stick_idx = env.world.cookbook.index["stick"]
    except Exception as e:
        raise RuntimeError("Failed to obtain the index for 'stick' from the cookbook.") from e
    if stick_idx is None:
        raise RuntimeError("The recipe database does not contain an entry for 'stick'.")

    # --------------------------------------------------------------------- #
    # 3. Breadth‑first search over the state‑space.
    # --------------------------------------------------------------------- #
    from collections import deque

    MAX_DEPTH = 80                     # generous but safe bound for small worlds
    visited = set()
    frontier = deque()
    frontier.append((start_state, []))

    while frontier:
        cur_state, actions = frontier.popleft()

        # Goal test: do we already have a stick?
        if cur_state.inventory[stick_idx] > 0:
            return actions

        # Depth pruning
        if len(actions) >= MAX_DEPTH:
            continue

        # Encode a hashable representation of the full state to avoid revisits.
        # ``tobytes`` gives a deterministic byte string for the numpy arrays.
        state_key = (
            cur_state.pos,
            cur_state.dir,
            cur_state.inventory.tobytes(),
            cur_state.grid.tobytes(),
        )
        if state_key in visited:
            continue
        visited.add(state_key)

        # Expand all possible actions (0‑4 as defined by CraftLab.action_specs)
        for a in range(5):
            # ``step`` returns (reward, new_state); the original state is left untouched.
            _, new_state = cur_state.step(a)
            frontier.append((new_state, actions + [a]))

    # --------------------------------------------------------------------- #
    # 4. If we exit the loop without finding a solution, return an empty plan.
    # --------------------------------------------------------------------- #
    return []

    # If the goal itself is a primitive, `primitives_for` returns an empty dict.

        # Take the first required primitive (index) – sufficient for a simple policy
        prim_idx = next(iter(needed))
    else:
        # Goal is already a primitive (e.g., "stick")
        prim_idx = goal_idx

    grid = state.grid                      # shape: (W, H, n_kinds)
    primitive_cells = np.argwhere(grid[:, :, prim_idx] > 0)

    if primitive_cells.size == 0:
        return []  # nothing to do

    # Choose the nearest primitive cell (Manhattan distance)
    agent_x, agent_y = state.pos
    dists = np.abs(primitive_cells[:, 0] - agent_x) + np.abs(primitive_cells[:, 1] - agent_y)
    target_idx = dists.argmin()
    target_x, target_y = primitive_cells[target_idx]

    # Build a sequence of moves towards the target cell
    actions = []

    # Move vertically
    while agent_x < target_x:
        actions.append(DOWN)
        agent_x += 1
    while agent_x > target_x:
        actions.append(UP)
        agent_x -= 1

    # Move horizontally
    while agent_y < target_y:
        actions.append(RIGHT)
        agent_y += 1
    while agent_y > target_y:
        actions.append(LEFT)
        agent_y -= 1

    # Finally, use the primitive (pick up / craft)
    actions.append(USE)

    return actions

        tx, ty = target
        actions = []

        # vertical movement
        while sx < tx:
            actions.append(ACTION_DOWN)
            sx += 1
        while sx > tx:
            actions.append(ACTION_UP)
            sx -= 1

        # horizontal movement
        while sy < ty:
            actions.append(ACTION_RIGHT)
            sy += 1
        while sy > ty:
            actions.append(ACTION_LEFT)
            sy -= 1

        return actions

    # ------------------------------------------------------------------
    actions = []
    cur_pos = pos

    return actions

    """Generate a list of actions that makes a stick in the given CraftLab environment.

    The function inspects the environment's recipe for ``stick``,
    collects all required primitive resources and finally crafts the
    stick at the appropriate workshop.  It uses a simple BFS planner
    on the current grid layout to move to adjacent cells of the target
    objects and issues a ``USE`` (action index 4) to pick‑up / craft.
    """
    import numpy as np
    from collections import deque

    # ------------------------------------------------------------------
    # Helper utilities
    # ------------------------------------------------------------------
    # Action indices (as defined in CraftLab.action_specs)
    ACTION_DOWN, ACTION_UP, ACTION_LEFT, ACTION_RIGHT, ACTION_USE = 0, 1, 2, 3, 4
    DIRS = [
        (1, 0),   # DOWN  -> action 0
        (-1, 0),  # UP    -> action 1
        (0, -1),  # LEFT  -> action 2
        (0, 1),   # RIGHT -> action 3
    ]

    def bfs(start, goal_cells, walkable):
        """Return a list of action indices that moves from ``start`` to any cell in ``goal_cells``."""
        w, h = walkable.shape
        q = deque()
        q.append((start, []))
        visited = {start}
        while q:
            (x, y), path = q.popleft()
            if (x, y) in goal_cells:
                return path
            for a_idx, (dx, dy) in enumerate(DIRS):
                nx, ny = x + dx, y + dy
                if 0 <= nx < w and 0 <= ny < h and walkable[nx, ny] and (nx, ny) not in visited:
                    visited.add((nx, ny))
                    q.append(((nx, ny), path + [a_idx]))
        # no path found – return empty list (caller will ignore)
        return []

    def move(pos, actions):
        """Apply a list of movement actions to a position."""
        x, y = pos
        for a in actions:
            dx, dy = DIRS[a]
            x, y = x + dx, y + dy
        return (x, y)

    # ------------------------------------------------------------------
    # Ensure the environment is reset and obtain the underlying state
    # ------------------------------------------------------------------
    if not hasattr(env, "_current_state") or env._current_state is None:
        # reset returns the initial observation; we ignore it here
        env.reset()
    state = env._current_state          # type: CraftState
    world = env.world                    # CraftWorld instance

    # ------------------------------------------------------------------
    # Determine what we need to collect for the goal (stick)
    # ------------------------------------------------------------------
    goal_idx = env.task.goal                     # integer index of the desired item
    recipe = world.cookbook.recipes.get(goal_idx, {})
    # primitives needed for one stick (dict {prim_idx: count})
    primitives_needed = world.cookbook.primitives_for(goal_idx)

    # ------------------------------------------------------------------
    # Grid information
    # ------------------------------------------------------------------
    grid = state.grid
    width, height, n_kinds = grid.shape
    # walkable cells are those that are completely empty (no entity present)
    walkable = np.sum(grid, axis=2) == 0

    actions_plan = []
    cur_pos = state.pos

    # ------------------------------------------------------------------
    # Collect each required primitive (respecting multiplicities)
    # ------------------------------------------------------------------
    for prim_idx, needed_cnt in primitives_needed.items():
        for _ in range(needed_cnt):
            # locate any remaining instance of this primitive on the grid
            prim_cells = np.argwhere(grid[:, :, prim_idx] > 0)
            if prim_cells.shape[0] == 0:
                # nothing left – break early
                break
            # pick the first occurrence (could be improved with nearest)
            px, py = prim_cells[0]

            # compute walkable neighbours of the primitive cell
            neighbour_cells = set()
            for dx, dy in DIRS:
                nx, ny = px + dx, py + dy
                if 0 <= nx < width and 0 <= ny < height and walkable[nx, ny]:
                    neighbour_cells.add((nx, ny))

            # If no walkable neighbour (unlikely), skip this primitive
            if not neighbour_cells:
                continue

            # plan a path to the nearest neighbour
            path = bfs(cur_pos, neighbour_cells, walkable)
            actions_plan.extend(path)
            cur_pos = move(cur_pos, path)

            # use the primitive (action 4)
            actions_plan.append(ACTION_USE)

            # Update our local copy of the grid so that the cell becomes empty.
            # This helps later planning (though not strictly required).
            grid[px, py, prim_idx] = 0
            walkable[px, py] = True

    # ------------------------------------------------------------------
    # Move to the workshop for the stick and craft it
    # ------------------------------------------------------------------
    workshop_idx = recipe.get("_at", None)
    if workshop_idx is not None:
        # locate a workshop cell
        workshop_cells = np.argwhere(grid[:, :, workshop_idx] > 0)
        if workshop_cells.shape[0] > 0:
            wx, wy = workshop_cells[0]

            # walkable neighbours of the workshop
            neighbour_cells = set()
            for dx, dy in DIRS:
                nx, ny = wx + dx, wy + dy
                if 0 <= nx < width and 0 <= ny < height and walkable[nx, ny]:
                    neighbour_cells.add((nx, ny))

            if neighbour_cells:
                path = bfs(cur_pos, neighbour_cells, walkable)
                actions_plan.extend(path)
                cur_pos = move(cur_pos, path)

                # finally craft the stick
                actions_plan.append(ACTION_USE)

    return actions_plan

        if len(actions) >= max_steps:

    return []

    Craft environment. The function performs a breadth‑first search over the
    deterministic state transition graph using the `CraftState.step` method,
    which returns a new immutable state for each action.  The search stops as

        achieve the goal.  If a plan cannot be found within the depth limit,
        an empty list is returned.

    # Helper to obtain the integer index for the primitive/item "stick".
    # ------------------------------------------------------------------
    try:
        stick_idx = env.world.cookbook.index["stick"]
    except Exception as e:
        raise RuntimeError("Unable to locate 'stick' in the cookbook index.") from e

    # Starting state (the environment should already be reset).
    start_state = env._current_state
    if start_state is None:
        raise RuntimeError("Environment does not have an initial state (reset not called?).")

    # ------------------------------------------------------------------
    # Bread‑first search over deterministic state transitions.
    # ------------------------------------------------------------------
    MAX_DEPTH = 30                     # reasonable bound for typical maps
    ACTION_SPACE = range(5)            # DOWN, UP, LEFT, RIGHT, USE

    queue = collections.deque()
    queue.append((start_state, []))

    # Simple visited set to avoid revisiting identical (pos, inventory) combos.
    visited = set()

    while queue:
        cur_state, actions = queue.popleft()

        # Goal test: do we already have a stick?
        if cur_state.inventory[stick_idx] > 0:
            return actions

        # Depth limit.
        if len(actions) >= MAX_DEPTH:
            continue

        # Visited check – we hash position + inventory (as a tuple of ints).
        key = (cur_state.pos, tuple(cur_state.inventory.tolist()))
        if key in visited:
            continue
        visited.add(key)

        # Expand all possible actions.
        for a in ACTION_SPACE:
            _, next_state = cur_state.step(a)   # deterministic transition
            queue.append((next_state, actions + [a]))

    # If we exit the loop no plan was found within the depth limit.
    return []

    # ------------------------------------------------------------------

            return []

        q = deque()
        q.append((start, []))
        visited = {start}
        while q:
            (x, y), path = q.popleft()
            for (dx, dy), act in [((1, 0), DOWN), ((-1, 0), UP),

                nx, ny = x + dx, y + dy
                if not (0 <= nx < W and 0 <= ny < H):
                    continue
                if (nx, ny) in visited:
                    continue
                # obstacle handling
                if occupied[nx, ny]:
                    if not (allow_target and (nx, ny) == goal):
                        continue
                visited.add((nx, ny))
                new_path = path + [act]
                if (nx, ny) == goal:
                    return new_path
                q.append(((nx, ny), new_path))
        # No path found – return empty list (should not happen in valid maps)
        return []

    # Gather required objects from the environment
    # ----------------------------------------------------------------------
    # current state
    state = env._current_state                     # CraftState instance

    goal_idx = env.task.goal

    primitives_needed = env.world.cookbook.primitives_for(goal_idx)  # {idx: count}

    # occupancy mask (True where any entity exists)
    occupied = grid.sum(axis=2) > 0

    actions = []

    # ----------------------------------------------------------------------
    # Collect each required primitive
    # ----------------------------------------------------------------------
    for prim_idx, needed_cnt in primitives_needed.items():
        for _ in range(needed_cnt):

            locs = np.argwhere(grid[:, :, prim_idx] > 0)
            if locs.size == 0:
                # no more primitives of this type – abort planning
                return actions
            target = tuple(locs[0])  # pick the first occurrence

            path = bfs(pos, target, occupied, allow_target=True)
            actions.extend(path)

            actions.append(USE)

            pos = target
            # remove the primitive from the grid (it is now in inventory)
            grid[target[0], target[1], prim_idx] = 0

            occupied[target[0], target[1]] = False

    initial grid, plans shortest Manhattan paths to an adjacent cell and
    issues ``USE`` actions to collect/craft.
    """
    import re
    from collections import deque
    import numpy as np

    # ----------------------------------------------------------------------
    # Helper utilities
    # ----------------------------------------------------------------------
    # Action mapping (as defined in the environment)
    ACTION_DOWN, ACTION_UP, ACTION_LEFT, ACTION_RIGHT, ACTION_USE = 0, 1, 2, 3, 4
    DIR_DELTAS = {
        ACTION_DOWN: (0, 1),
        ACTION_UP: (0, -1),
        ACTION_LEFT: (-1, 0),
        ACTION_RIGHT: (1, 0),
    }

    def bfs(start, goals, blocked):
        """Return a list of actions from *start* to the nearest *goals* cell.

        *blocked* is a 2‑D boolean array where True means the cell cannot be
        entered.  *goals* is a set of (x, y) tuples.  If no path is found,
        an empty list is returned.
        """
        w, h = blocked.shape
        sx, sy = start
        if (sx, sy) in goals:
            return []  # already at a goal cell
        visited = np.full((w, h), False, dtype=bool)
        prev = dict()  # (x,y) -> (prev_x, prev_y, action)

        q = deque()
        q.append((sx, sy))
        visited[sx, sy] = True

        while q:
            x, y = q.popleft()
            for act, (dx, dy) in DIR_DELTAS.items():
                nx, ny = x + dx, y + dy
                if 0 <= nx < w and 0 <= ny < h and not visited[nx, ny] and not blocked[nx, ny]:
                    visited[nx, ny] = True
                    prev[(nx, ny)] = (x, y, act)
                    if (nx, ny) in goals:
                        # reconstruct path
                        actions = []
                        cx, cy = nx, ny
                        while (cx, cy) != (sx, sy):
                            px, py, a = prev[(cx, cy)]
                            actions.append(a)
                            cx, cy = px, py
                        actions.reverse()
                        return actions
                    q.append((nx, ny))
        return []  # no path found

    # ----------------------------------------------------------------------
    # Initialise / reset environment to obtain the initial state
    # ----------------------------------------------------------------------
    # Reset returns an observation but also creates the internal state.
    env.reset()
    state = env._current_state  # type: CraftState
    grid = state.grid.copy()    # (W, H, n_kinds)

    # ----------------------------------------------------------------------
    # Parse the goal item name from the task description
    # ----------------------------------------------------------------------
    # The task name is something like "make[stick]"
    goal_expr = getattr(env.task, "goal", None)
    if goal_expr is None:
        # fallback – sometimes the goal may be stored as a string directly
        goal_expr = getattr(env, "task_name", "")
    # Extract the identifier inside brackets, otherwise use the whole string
    m = re.search(r"\[(\w+)\]", str(goal_expr))
    goal_name = m.group(1) if m else str(goal_expr)

    # ----------------------------------------------------------------------
    # Resolve indices using the cookbook
    # ----------------------------------------------------------------------
    cookbook = env.world.cookbook
    goal_idx = cookbook.index[goal_name]

    # Determine required primitive resources for the goal.
    # If the goal itself is primitive we just need to collect it.
    primitives_needed = {}
    if goal_idx in cookbook.primitives:
        primitives_needed[goal_idx] = 1
    else:
        # Obtain ingredient list from the recipe (excluding the "_at" key)
        recipe = cookbook.recipes.get(goal_idx, {})
        for ing_idx, cnt in recipe.items():
            if ing_idx == "_at":
                continue
            primitives_needed[ing_idx] = primitives_needed.get(ing_idx, 0) + cnt

    # ----------------------------------------------------------------------
    # Prepare a mask of blocked cells (any non‑empty cell is considered blocked)
    # ----------------------------------------------------------------------
    # A cell is blocked if any kind is present (including other primitives)
    blocked_static = (grid.sum(axis=2) > 0)  # shape (W, H)

    # ----------------------------------------------------------------------
    # Planning loop – collect each required primitive
    # ----------------------------------------------------------------------
    actions_plan = []
    cur_pos = state.pos  # (x, y)

    for kind_idx, needed_cnt in primitives_needed.items():
        for _ in range(needed_cnt):
            # Locate a remaining instance of the required primitive
            locations = np.argwhere(grid[:, :, kind_idx] > 0)
            if locations.size == 0:
                # No more instances – abort planning
                break
            # Choose the first found location (could be improved with nearest)
            tx, ty = locations[0]

            # Compute the set of admissible adjacent cells where we can stand
            adj_cells = set()
            for dx, dy in DIR_DELTAS.values():
                ax, ay = tx + dx, ty + dy
                if 0 <= ax < grid.shape[0] and 0 <= ay < grid.shape[1]:
                    # The adjacent cell must be free (not blocked)
                    if not blocked_static[ax, ay]:
                        adj_cells.add((ax, ay))
            # If we cannot stand adjacent (maybe surrounded), also allow standing
            # on the primitive cell itself (some maps allow that).
            if not adj_cells and not blocked_static[tx, ty]:
                adj_cells.add((tx, ty))

            # Find shortest path to any of the adjacent cells
            path = bfs(cur_pos, adj_cells, blocked_static)
            actions_plan.extend(path)

            # Update current position according to the executed moves
            for act in path:
                dx, dy = DIR_DELTAS[act]
                cur_pos = (cur_pos[0] + dx, cur_pos[1] + dy)

            # Issue the USE action to pick up the primitive
            actions_plan.append(ACTION_USE)
            # Simulate the pickup in our local copy:
            grid[tx, ty, kind_idx] = 0
            # After picking up, the cell becomes empty – update blocked mask
            blocked_static[tx, ty] = False

            # Position does not change after USE
            # (agent stays on the same cell)

    # ----------------------------------------------------------------------
    # If the goal is not primitive, move to the required workshop and craft
    # ----------------------------------------------------------------------
    if goal_idx not in cookbook.primitives:
        recipe = cookbook.recipes.get(goal_idx, {})
        workshop_name = recipe.get("_at")
        if workshop_name:
            workshop_idx = cookbook.index[workshop_name]
            # Locate a workshop cell
            ws = np.argwhere(grid[:, :, workshop_idx] > 0)
            if ws.size > 0:
                wx, wy = ws[0]
                # Adjacent cells to the workshop (must be free)
                adj_cells = set()
                for dx, dy in DIR_DELTAS.values():
                    ax, ay = wx + dx, wy + dy
                    if 0 <= ax < grid.shape[0] and 0 <= ay < grid.shape[1]:
                        if not blocked_static[ax, ay]:
                            adj_cells.add((ax, ay))
                # Fallback to standing on the workshop cell if free
                if not adj_cells and not blocked_static[wx, wy]:
                    adj_cells.add((wx, wy))

                path = bfs(cur_pos, adj_cells, blocked_static)
                actions_plan.extend(path)
                for act in path:
                    dx, dy = DIR_DELTAS[act]
                    cur_pos = (cur_pos[0] + dx, cur_pos[1] + dy)

                # Use the workshop to craft the goal item
                actions_plan.append(ACTION_USE)

    # Return the complete list of action indices
    return actions_plan

    Simple planner that moves the agent to a free cell adjacent to a stick,
    faces the stick and issues a USE action. Returns a list of action
    indices (0=DOWN, 1=UP, 2=LEFT, 3=RIGHT, 4=USE).
    """
    # access the underlying state
    state = getattr(env, "_current_state", None)
    if state is None:
        return []

    # shortcut: if we already have a stick in inventory, nothing to do
    world = env.world
    idx = world.cookbook.index
    if "stick" not in idx.contents:
        return []
    stick_idx = idx["stick"]
    if state.inventory[stick_idx] > 0:
        return []

    grid = state.grid                     # shape (W, H, K)
    W, H, _ = grid.shape

    # locate a stick on the ground
    stick_cells = np.argwhere(grid[:, :, stick_idx] > 0)
    if len(stick_cells) == 0:
        # no stick present – nothing we can do with this simple planner
        return []

    sx, sy = stick_cells[0]               # pick the first occurrence

    # find a free neighbour cell to stand on
    neighbours = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    target = None
    for dx, dy in neighbours:
        ax, ay = sx + dx, sy + dy
        if 0 <= ax < W and 0 <= ay < H and not grid[ax, ay].any():
            target = (ax, ay)
            break
    if target is None:
        # cannot reach the stick (surrounded)
        return []

    # ------------------------------------------------------------------
    # BFS to find a path from current position to the target cell
    # ------------------------------------------------------------------
    start = state.pos                     # (x, y)
    from collections import deque

    q = deque([start])
    visited = {start}
    # store predecessor and the action used to get to a node
    pred = {}

    # mapping of movement deltas to action indices
    moves = [
        (0, 1, 0),   # DOWN
        (0, -1, 1),  # UP
        (-1, 0, 2),  # LEFT
        (1, 0, 3),   # RIGHT
    ]

    while q:
        x, y = q.popleft()
        if (x, y) == target:
            break
        for dx, dy, act in moves:
            nx, ny = x + dx, y + dy
            if 0 <= nx < W and 0 <= ny < H and not grid[nx, ny].any():
                if (nx, ny) not in visited:
                    visited.add((nx, ny))
                    pred[(nx, ny)] = ((x, y), act)
                    q.append((nx, ny))

    # reconstruct action sequence
    if target not in pred and start != target:
        # no path found
        return []

    actions = []
    cur = target
    while cur != start:
        (prev_node, act) = pred[cur]
        actions.append(act)
        cur = prev_node
    actions = list(reversed(actions))

    # ------------------------------------------------------------------
    # Face the stick and use it
    # ------------------------------------------------------------------
    ax, ay = target
    if sx > ax:
        face_action = 3          # RIGHT
    elif sx < ax:
        face_action = 2          # LEFT
    elif sy > ay:
        face_action = 0          # DOWN
    elif sy < ay:
        face_action = 1          # UP
    else:
        face_action = None

    if face_action is not None:
        actions.append(face_action)

    actions.append(4)   # USE

    return actions

    # 4.  Helper utilities.
    # --------------------------------------------------------------------- #

        """Return a list of actions moving from cur → target (Manhattan)."""
        cx, cy = cur
        tx, ty = target
        actions = []

        # move in x‑direction
        while cx < tx:
            actions.append(RIGHT)
            cx += 1
        while cx > tx:
            actions.append(LEFT)
            cx -= 1
        # move in y‑direction
        while cy < ty:
            actions.append(DOWN)
            cy += 1
        while cy > ty:
            actions.append(UP)
            cy -= 1
        return actions, (cx, cy)

        """Find an empty neighbour of (x, y); returns (nx, ny) or None."""
        h, w, _ = grid.shape
        candidates = [(x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)]
        for nx, ny in candidates:
            if 0 <= nx < h and 0 <= ny < w:
                if grid[nx, ny, :].sum() == 0:
                    return (nx, ny)
        return None

    # 5.  Build the action list.
    # --------------------------------------------------------------------- #

    workshop.

    The function works directly with the internal objects of a
    ``CraftLab`` environment:

      4. After all primitives are collected, move to the workshop specified by

    # --------------------------------------------------------------------- #
    def move_towards(pos, target):

        x, y = pos
        tx, ty = target
        if x < tx:
            return 3          # RIGHT
        if x > tx:
            return 2          # LEFT
        if y < ty:
            return 0          # DOWN
        if y > ty:
            return 1          # UP
        return None          # already there

        x, y = pos
        if action == 0 and y + 1 < h:   # DOWN
            return (x, y + 1)
        if action == 1 and y - 1 >= 0:  # UP
            return (x, y - 1)
        if action == 2 and x - 1 >= 0:  # LEFT
            return (x - 1, y)
        if action == 3 and x + 1 < w:   # RIGHT
            return (x + 1, y)
        return pos                      # blocked or out of bounds – stay

    # --------------------------------------------------------------------- #
    # Extract world information
    # --------------------------------------------------------------------- #
    state = env._current_state                # current CraftState
    grid = state.grid
    width, height, _ = grid.shape
    pos = tuple(state.pos)                    # (x, y) tuple

    goal_idx = env.task.goal                   # integer index of ``stick``
    cookbook = env.world.cookbook

    # --------------------------------------------------------------------- #
    # 1. Determine needed primitives for the goal
    # --------------------------------------------------------------------- #
    needed = cookbook.primitives_for(goal_idx)   # dict {prim_idx: count}
    actions = []

    # --------------------------------------------------------------------- #
    # 2. Collect each required primitive
    # --------------------------------------------------------------------- #
    for prim_idx, count in needed.items():
        for _ in range(count):
            # locate the first cell that contains the primitive
            locations = list(zip(*np.where(grid[:, :, prim_idx] > 0)))
            if not locations:

                break
            target = locations[0]  # (x, y) position of the primitive

            # Move towards the primitive cell
            while pos != target:
                act = move_towards(pos, target)
                if act is None:
                    break
                actions.append(act)
                pos = apply_move(pos, act, width, height)

            # Use the primitive (collect / interact)
            actions.append(4)          # USE
            # Assume the environment will remove the primitive; we keep the
            # simulated position unchanged.

    # --------------------------------------------------------------------- #
    # 3. Move to the appropriate workshop and craft the item
    # --------------------------------------------------------------------- #
    recipe = cookbook.recipes.get(goal_idx, {})
    workshop_idx = recipe.get('_at')

                if act is None:
                    break
                actions.append(act)
                pos = apply_move(pos, act, width, height)
            # final USE to craft the item
            actions.append(4)          # USE

    return actions

            grid = state.grid

            state = env._current_state
            actions.append(4)

    # --------------------------------------------------------------------- #

    implementation is safe because the surrounding code does not rely
    on any side‑effects from this function; it only expects a list of
    integer actions.
    """
    return []

0,0.0,def make_stick(env):
    """
    Generate a sequence of actions that moves the agent next to a stick resource
    and uses it. The function works for the typical CraftWorld setup where
    `stick` is a primitive that can be collected directly from the grid.
    """
    # Ensure the environment is reset and we have access to the internal state
    try:
        # If the environment has already been reset, just use the current state
        state = env._current_state
    except AttributeError:
        # Reset returns an observation dict; the internal state is stored inside env
        env.reset()
        state = env._current_state

    # Shortcut references
    world = env.world                     # CraftWorld instance
    grid = state.grid                     # (W, H, n_kinds) occupancy array
    pos = tuple(state.pos)                # current (x, y) integer position

    # Identify the integer index for the item "stick"
    stick_idx = world.cookbook.index["stick"]

    # Locate a stick cell on the grid (first occurrence)
    stick_cells = list(zip(*np.where(grid[:, :, stick_idx] > 0)))
    if not stick_cells:
        # No stick found – nothing to do
        return []

    tx, ty = stick_cells[0]  # target stick coordinates

    # Determine which neighboring cell we can stand on (must be empty)
    W, H = grid.shape[0], grid.shape[1]
    occupied = grid.sum(axis=2) > 0  # True where any entity occupies the cell

    neighbor_dirs = [(-1, 0), (1, 0), (0, -1), (0, 1)]  # left, right, up, down
    neighbor = None
    for dx, dy in neighbor_dirs:
        nx, ny = tx + dx, ty + dy
        if 0 <= nx < W and 0 <= ny < H and not occupied[nx, ny]:
            neighbor = (nx, ny)
            break
    if neighbor is None:
        # No free adjacent cell; cannot collect the stick
        return []

    # ----------------------------------------------------------------------
    # BFS to find a shortest path from the current position to the neighbor
    # ----------------------------------------------------------------------
    from collections import deque

    def bfs(start, goal):
        q = deque([start])
        prev = {start: None}
        while q:
            cur = q.popleft()
            if cur == goal:
                break
            cx, cy = cur
            for dx, dy in neighbor_dirs:
                nx, ny = cx + dx, cy + dy
                if 0 <= nx < W and 0 <= ny < H and not occupied[nx, ny]:
                    nxt = (nx, ny)
                    if nxt not in prev:
                        prev[nxt] = cur
                        q.append(nxt)
        # Reconstruct path (list of positions, excluding the start)
        if goal not in prev:
            return []  # unreachable
        path = []
        cur = goal
        while prev[cur] is not None:
            path.append(cur)
            cur = prev[cur]
        path.reverse()
        return path

    path = bfs(pos, neighbor)

    # Convert the path positions into action indices
    action_map = {
        (0, 1): 0,   # DOWN
        (0, -1): 1,  # UP
        (-1, 0): 2,  # LEFT
        (1, 0): 3,   # RIGHT
    }
    actions = []
    cur_pos = pos
    for nxt in path:
        dx = nxt[0] - cur_pos[0]
        dy = nxt[1] - cur_pos[1]
        act = action_map.get((dx, dy))
        if act is None:
            # Should never happen with valid moves
            continue
        actions.append(act)
        cur_pos = nxt

    # Finally, use the stick (action index 4)
    actions.append(4)  # USE

    return actions
