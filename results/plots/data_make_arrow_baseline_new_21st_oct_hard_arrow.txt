    """
    Generate a sequence of actions that collects the required primitives for the
    Arrow recipe and then crafts the Arrow at its workshop.

    This implementation works for the standard CraftWorld environments where
    the world layout is fully observable via the internal state of the
    environment. It extracts the goal index, determines which primitive items are
    needed, locates them on the grid, moves to each of them, uses the `USE`
    action to pick them up, then moves to the appropriate workshop and uses the
    `USE` action again to craft the final item.

    The action indices follow the mapping defined in CraftLab:
        DOWN = 0, UP = 1, LEFT = 2, RIGHT = 3, USE = 4
    """
    import numpy as np

    # ----- helpers ---------------------------------------------------------
    DOWN, UP, LEFT, RIGHT, USE = 0, 1, 2, 3, 4

    def move_to(target, cur_pos, action_list):
        """Append actions that move the agent from cur_pos to target (x, y)."""
        x, y = cur_pos
        tx, ty = target
        # horizontal moves
        while x < tx:
            action_list.append(RIGHT)
            x += 1
        while x > tx:
            action_list.append(LEFT)
            x -= 1
        # vertical moves
        while y < ty:
            action_list.append(DOWN)
            y += 1
        while y > ty:
            action_list.append(UP)
            y -= 1
        return (x, y)

    # ----- extract current information ------------------------------------
    state = env._current_state                       # current CraftState
    goal_idx = env.task.goal                         # index of the desired item
    cookbook = env.world.cookbook

    # primitives needed for the goal (primitive_idx -> required count)
    needed_prims = cookbook.primitives_for(goal_idx)

    actions = []
    cur_pos = list(state.pos)  # mutable copy of (x, y)

    # ----- collect each required primitive ---------------------------------
    for prim_idx, count in needed_prims.items():
        # work on a mutable copy of the slice that tells where this primitive lies
        prim_grid = np.copy(state.grid[:, :, prim_idx])

        for _ in range(count):
            # locate all cells that still contain this primitive
            locations = np.argwhere(prim_grid > 0)
            if locations.size == 0:
                # no more of this primitive on the map – stop trying
                break

            # pick the nearest one (Manhattan distance)
            distances = np.abs(locations[:, 0] - cur_pos[0]) + np.abs(locations[:, 1] - cur_pos[1])
            nearest_idx = np.argmin(distances)
            target = tuple(locations[nearest_idx])

            # move to the primitive and pick it up
            cur_pos = move_to(target, cur_pos, actions)
            actions.append(USE)

            # update our local copy of the grid (primitive has been taken)
            prim_grid[target[0], target[1]] -= 1

    # ----- go to the workshop and craft ------------------------------------
    recipe = cookbook.recipes.get(goal_idx, {})
    workshop_idx = recipe.get('_at')
    if workshop_idx is not None:
        # find any workshop cell of the required type
        workshop_grid = state.grid[:, :, workshop_idx]
        workshop_locs = np.argwhere(workshop_grid > 0)
        if workshop_locs.size > 0:
            # choose the nearest workshop
            distances = np.abs(workshop_locs[:, 0] - cur_pos[0]) + np.abs(workshop_locs[:, 1] - cur_pos[1])
            nearest_idx = np.argmin(distances)
            workshop_target = tuple(workshop_locs[nearest_idx])

            # move to the workshop and use it to craft
            cur_pos = move_to(workshop_target, cur_pos, actions)
            actions.append(USE)

    return actions

    """
    Produce a list of actions that will collect the required primitives for an
    ``arrow`` and then craft it at the appropriate workshop.

    The function works with a ``CraftLab`` instance (the ``env`` argument).
    It queries the world’s cookbook for the primitive requirements of the
    current goal, navigates to each primitive, uses ``USE`` to pick it up,
    and finally moves to the workshop indicated by the recipe and uses
    ``USE`` again to craft the item.
    """
    import numpy as np
    from collections import deque

    # ----------------------------------------------------------------------
    # Helper: shortest path (Manhattan) avoiding occupied cells.
    # Returns a list of movement actions (ints) that brings the agent from
    # ``start`` to any cell in ``targets`` (all targets must be free cells).
    # ----------------------------------------------------------------------
    def bfs_path(start, targets, grid, move_actions):
        W, H, _ = grid.shape
        visited = set()
        q = deque()
        q.append((start, []))
        visited.add(start)

        while q:
            (x, y), path = q.popleft()
            if (x, y) in targets:
                return path

            for name, (dx, dy) in [
                ("UP", (0, -1)),
                ("DOWN", (0, 1)),
                ("LEFT", (-1, 0)),
                ("RIGHT", (1, 0)),
            ]:
                nx, ny = x + dx, y + dy
                if 0 <= nx < W and 0 <= ny < H and (nx, ny) not in visited:
                    # a cell is passable iff it contains no entities
                    if not np.any(grid[nx, ny, :]):
                        visited.add((nx, ny))
                        q.append(((nx, ny), path + [move_actions[name]]))
        return []  # fallback – should not happen in well‑formed maps

    # ----------------------------------------------------------------------
    # Retrieve useful information from the environment
    # ----------------------------------------------------------------------
    # action indices
    act_map = env.action_specs()
    USE = act_map["USE"]
    move_actions = {
        "UP": act_map["UP"],
        "DOWN": act_map["DOWN"],
        "LEFT": act_map["LEFT"],
        "RIGHT": act_map["RIGHT"],
    }

    # goal index (the item we have to craft)
    goal_idx = env.task.goal

    # primitive requirements for the goal
    primitives_needed = env.world.cookbook.primitives_for(goal_idx)

    actions = []

    # ----------------------------------------------------------------------
    # Collect each required primitive (respecting the required count)
    # ----------------------------------------------------------------------
    for prim_idx, cnt in primitives_needed.items():
        for _ in range(cnt):
            # current state (grid, position)
            state = env._current_state
            grid = state.grid
            pos = state.pos

            # locate a cell that still contains the primitive
            prim_cells = np.argwhere(grid[:, :, prim_idx] > 0)
            if prim_cells.size == 0:
                # already in inventory – nothing to do
                continue
            prim_x, prim_y = prim_cells[0][:2]

            # find free neighbour cells of the primitive (where we can stand)
            neighbour_targets = set()
            for dx, dy in [(0, -1), (0, 1), (-1, 0), (1, 0)]:
                nx, ny = prim_x + dx, prim_y + dy
                if 0 <= nx < grid.shape[0] and 0 <= ny < grid.shape[1]:
                    if not np.any(grid[nx, ny, :]):  # free cell
                        neighbour_targets.add((nx, ny))

            # compute a path to one of those free neighbours
            path = bfs_path(pos, neighbour_targets, grid, move_actions)
            actions.extend(path)

            # finally use the primitive
            actions.append(USE)

    # ----------------------------------------------------------------------
    # Craft the item at its workshop
    # ----------------------------------------------------------------------
    recipe = env.world.cookbook.recipes.get(goal_idx, {})
    workshop_kind = recipe.get("_at")
    if workshop_kind is not None:
        state = env._current_state
        grid = state.grid
        pos = state.pos

        # locate a workshop of the required kind
        workshop_cells = np.argwhere(grid[:, :, workshop_kind] > 0)
        if workshop_cells.size > 0:
            w_x, w_y = workshop_cells[0][:2]

            # free neighbours of the workshop
            neighbour_targets = set()
            for dx, dy in [(0, -1), (0, 1), (-1, 0), (1, 0)]:
                nx, ny = w_x + dx, w_y + dy
                if 0 <= nx < grid.shape[0] and 0 <= ny < grid.shape[1]:
                    if not np.any(grid[nx, ny, :]):
                        neighbour_targets.add((nx, ny))

            # path to a free neighbour
            path = bfs_path(pos, neighbour_targets, grid, move_actions)
            actions.extend(path)

            # use the workshop to craft the goal item
            actions.append(USE)

    return actions

    1. Resets the environment (so we have a fresh state).
    2. Determines which primitive items are required to craft an ``arrow``.
    3. Walks to each primitive, picks it up with the ``USE`` action.
    4. Walks to the required workshop and crafts the arrow with ``USE``.

    # ------------------------------------------------------------------
    # Helper to translate a Manhattan move into action indices.
    #   0: DOWN, 1: UP, 2: LEFT, 3: RIGHT, 4: USE
    # ------------------------------------------------------------------
    def _move_actions(src, dst):
        """Return a list of actions moving from src (y,x) to dst (y,x)."""
        actions = []
        dy = dst[0] - src[0]
        dx = dst[1] - src[1]

        # vertical movement first
        if dy > 0:
            actions.extend([0] * dy)          # DOWN
        elif dy < 0:
            actions.extend([1] * (-dy))       # UP

        # then horizontal movement
        if dx > 0:
            actions.extend([3] * dx)          # RIGHT
        elif dx < 0:
            actions.extend([2] * (-dx))       # LEFT
        return actions

    # ------------------------------------------------------------------
    # 1. Reset environment to obtain a fresh state.
    # ------------------------------------------------------------------
    # ``env.reset`` returns an observation dict; the internal state is stored
    # in ``env._current_state`` (the CraftLab implementation does this).
    env.reset()
    state = env._current_state          # type: CraftState
    world = env.world                    # reference to CraftWorld

    # ------------------------------------------------------------------
    # 2. Identify the goal (arrow) and its primitive requirements.
    # ------------------------------------------------------------------
    goal_name = "arrow"
    goal_idx = world.cookbook.index[goal_name]           # integer ID of arrow
    primitive_requirements = world.cookbook.primitives_for(goal_idx)
    # ``primitive_requirements`` -> {prim_idx: count, ...}

    actions_plan = []

    # ------------------------------------------------------------------
    # 3. Collect each required primitive.
    # ------------------------------------------------------------------
    for prim_idx, needed_cnt in primitive_requirements.items():
        for _ in range(needed_cnt):
            # Find all cells that still contain the primitive.
            prim_cells = np.argwhere(state.grid[:, :, prim_idx] > 0)
            if prim_cells.size == 0:
                # No more primitive cells – break (should not happen in a
                # correctly generated scenario).
                break

            # Choose the first available primitive cell (could be improved to pick
            # the nearest, but any works for the deterministic test environments).
            target = tuple(prim_cells[0])

            # Move to the primitive cell.
            actions_plan.extend(_move_actions(state.pos, target))

            # Pick it up.
            actions_plan.append(4)   # USE

            # Update our internal copy of the state so subsequent moves are based
            # on the new position.
            state.pos = target
            # Simulate the effect of picking up the item.
            state.grid[target[0], target[1], prim_idx] -= 1
            state.inventory[prim_idx] += 1

    # ------------------------------------------------------------------
    # 4. Go to the appropriate workshop and craft the arrow.
    # ------------------------------------------------------------------
    # The recipe dictionary stores the workshop name under the special key
    # ``'_at'``.
    workshop_name = world.cookbook.recipes[goal_idx]["_at"]
    workshop_idx = world.cookbook.index[workshop_name]

    workshop_cells = np.argwhere(state.grid[:, :, workshop_idx] > 0)
    if workshop_cells.size > 0:
        workshop_pos = tuple(workshop_cells[0])
        actions_plan.extend(_move_actions(state.pos, workshop_pos))
        actions_plan.append(4)   # USE to craft

    return actions_plan

    """
    Generate a deterministic action sequence that collects the required primitives
    for an ``arrow`` and then crafts it at the appropriate workshop.

    The function works for any CraftWorld environment where the goal is the
    ``arrow`` item.  It inspects the environment’s internal state to discover:
      * the index of the goal item,
      * which primitive kinds (and how many of each) are needed,
      * the locations of those primitives on the grid,
      * the location(s) of the workshop required for the recipe.

    It then builds a simple Manhattan‑distance plan:
      1. Move to each needed primitive, ``USE`` to pick it up.
      2. Move to the nearest workshop and ``USE`` to craft the arrow.

    The returned list contains integer actions according to the mapping used
    by ``CraftLab``:
        DOWN → 0, UP → 1, LEFT → 2, RIGHT → 3, USE → 4
    """
    # Helper to generate movement actions from a start to a target coordinate.
    def _move_to(start, target):
        sx, sy = start
        tx, ty = target
        actions = []
        # X‑axis (RIGHT / LEFT)
        if tx > sx:
            actions += [3] * (tx - sx)   # RIGHT
        elif tx < sx:
            actions += [2] * (sx - tx)   # LEFT
        # Y‑axis (DOWN / UP)
        if ty > sy:
            actions += [0] * (ty - sy)   # DOWN
        elif ty < sy:
            actions += [1] * (sy - ty)   # UP
        return actions

    # --------------------------------------------------------------------- #
    # Extract the current state and recipe information from the environment.
    # --------------------------------------------------------------------- #
    state = env._current_state               # CraftState instance
    world = state.world                       # CraftWorld instance
    cookbook = world.cookbook                 # Cookbook with recipes

    # Goal index for the current task (arrow)
    goal_idx = env.task.goal

    # Determine which primitive kinds are required and how many of each.
    primitives_needed = cookbook.primitives_for(goal_idx)   # {kind_idx: count}
    # Identify the workshop kind required for this recipe (if any).
    workshop_kind = None
    if goal_idx in cookbook.recipes:
        workshop_kind = cookbook.recipes[goal_idx].get('_key')

    # --------------------------------------------------------------------- #
    # Locate all instances of each required primitive on the grid.
    # --------------------------------------------------------------------- #
    grid = state.grid                               # (W, H, n_kinds) array
    primitive_positions = {}
    for kind, cnt in primitives_needed.items():
        # Find every cell that contains this kind (value > 0).
        inds = np.argwhere(grid[:, :, kind] > 0)
        primitive_positions[kind] = inds.tolist()   # list of (x, y) pairs

    # Locate workshop cells (if a specific workshop is required).
    workshop_positions = []
    if workshop_kind is not None:
        inds = np.argwhere(grid[:, :, workshop_kind] > 0)
        workshop_positions = inds.tolist()

    # --------------------------------------------------------------------- #
    # Build the action plan.
    # --------------------------------------------------------------------- #
    actions = []
    cur_pos = state.pos  # current (x, y) location of the agent

    # Collect each needed primitive.
    for kind, needed_cnt in primitives_needed.items():
        available = primitive_positions.get(kind, [])
        # Take the first `needed_cnt` positions (should be enough in a valid env).
        for i in range(min(needed_cnt, len(available))):
            target = tuple(available[i])
            actions += _move_to(cur_pos, target)
            actions.append(4)            # USE to pick up the primitive
            cur_pos = target

    # Move to a workshop (if defined) and craft the arrow.
    if workshop_positions:
        # Choose the nearest workshop to the current location.
        nearest = min(

            key=lambda p: abs(p[0] - cur_pos[0]) + abs(p[1] - cur_pos[1])
        )
        actions += _move_to(cur_pos, tuple(nearest))
        actions.append(4)                # USE to craft at the workshop
        cur_pos = tuple(nearest)
    else:
        # No specific workshop – simply USE at the current location.
        actions.append(4)

    return actions

    """
    Simple planner that collects the required primitives for an ``arrow`` and
    then crafts it at the appropriate workshop.

    The function works with the CraftLab environment returned by
    ``EnvironmentFactory.sample_environment``.  It assumes the standard action
    mapping used by the environment:

        0 – DOWN
        1 – UP
        2 – LEFT
        3 – RIGHT
        4 – USE (collect / craft)

    The planner:
        1. Resets the environment to obtain the initial state.
        2. Determines the goal index (the item to be crafted) and the primitive
           requirements for that goal using the ``Cookbook``.
        3. Repeatedly moves to the closest cell that contains a needed primitive,
           issues a ``USE`` action to collect it, and updates the internal grid
           representation.
        4. After all primitives are collected, moves to the workshop required for
           the goal and issues a final ``USE`` to craft the item.

    The returned list contains the sequence of action integers that, when fed
    to ``env.step``, should complete the task.
    """
    import numpy as np
    from collections import deque

    # --------------------------------------------------------------------- #
    # Helper functions
    # --------------------------------------------------------------------- #
    def bfs(start, target, grid, blocked_kinds):
        """Bread‑first search returning a list of actions from start to target."""
        W, H = grid.shape[0], grid.shape[1]
        q = deque()
        q.append((start, []))
        visited = {start}
        # (dx, dy, action) – mapping matches the environment's action IDs
        moves = [(1, 0, 0),   # DOWN
                 (-1, 0, 1),  # UP
                 (0, -1, 2),  # LEFT
                 (0, 1, 3)]   # RIGHT
        while q:
            (x, y), path = q.popleft()
            if (x, y) == target:
                return path
            for dx, dy, act in moves:
                nx, ny = x + dx, y + dy
                if not (0 <= nx < W and 0 <= ny < H):
                    continue
                if (nx, ny) in visited:
                    continue
                # cell is blocked if any non‑grabbable entity is present
                if any(grid[nx, ny, k] > 0 for k in blocked_kinds):
                    continue
                visited.add((nx, ny))
                q.append(((nx, ny), path + [act]))
        # No path found
        return []

    def positions_of(kind, grid):
        """Return a list of (x, y) tuples where *kind* is present in *grid*."""
        idx = np.argwhere(grid[:, :, kind] > 0)
        return [tuple(p) for p in idx]

    # --------------------------------------------------------------------- #
    # Initialise environment and obtain static information
    # --------------------------------------------------------------------- #
    env.reset()                                 # start a fresh episode
    state = env._current_state                  # internal CraftState object
    pos = tuple(state.pos)                      # current agent position (x, y)
    grid = state.grid.copy()                    # mutable copy of the world grid
    world = env.world

    goal_idx = env.task.goal                     # integer index of the target item
    # Mapping from primitive kind index → how many are required
    primitive_needs = world.cookbook.primitives_for(goal_idx)
    # Set of entity kinds that are obstacles (cannot be traversed)
    blocked_kinds = world.non_grabbable_indices

    # --------------------------------------------------------------------- #
    # Collect each required primitive
    # --------------------------------------------------------------------- #
    action_seq = []
    for prim_kind, needed in primitive_needs.items():
        while needed > 0:
            # Locate all remaining cells that contain this primitive
            cand_cells = positions_of(prim_kind, grid)
            if not cand_cells:
                # No more of this primitive on the map – abort planning
                break

            # Choose the closest cell (Manhattan distance) as the next target
            distances = [abs(px - pos[0]) + abs(py - pos[1]) for (px, py) in cand_cells]
            target = cand_cells[int(np.argmin(distances))]

            # Move to the target cell
            move_actions = bfs(pos, target, grid, blocked_kinds)
            action_seq.extend(move_actions)
            pos = target

            # Collect the primitive
            action_seq.append(4)                 # USE
            needed -= 1

            # Update our internal grid copy – the primitive disappears after collection
            grid[pos[0], pos[1], prim_kind] = 0

    # --------------------------------------------------------------------- #
    # Move to the appropriate workshop and craft the goal item
    # --------------------------------------------------------------------- #
    # The workshop kind is stored under the special key "_at" in the recipe dict
    workshop_kind = world.cookbook.recipes.get(goal_idx, {}).get('_at')
    if workshop_kind is not None:
        workshop_cells = positions_of(workshop_kind, grid)
        if workshop_cells:
            distances = [abs(px - pos[0]) + abs(py - pos[1]) for (px, py) in workshop_cells]
            target_ws = workshop_cells[int(np.argmin(distances))]
            move_actions = bfs(pos, target_ws, grid, blocked_kinds)
            action_seq.extend(move_actions)
            pos = target_ws

            # Final USE to craft the item
            action_seq.append(4)                 # USE

    return action_seq

        dx, dy = ACTIONS[action]
        new_pos = (cur_pos[0] + dx, cur_pos[1] + dy)
        if in_bounds(new_pos, shape):
            cur_pos = new_pos
        cur_dir = action  # moving action also sets the facing direction

        actions.append(USE)   # craft at the workshop

    return actions

            move_to((int(wx), int(wy)))

    # 6. Return the generated action list
    # --------------------------------------------------------------------- #
    return actions
