96,1.5,def make_stick(env):
    """
    Simple stochastic planner for the ``make[stick]`` task.

    The function resets the environment, then performs random actions
    (sampling uniformly from the action space) until the task is
    completed or a safety limit of steps is reached.  All taken actions
    are recorded and the environment is reset a second time so that the
    caller (e.g. ``solve``) receives a fresh environment on which it can
    replay the returned action list.

    Parameters
    ----------
    env : CraftLab (or compatible) instance
        The environment on which to plan actions.

    Returns
    -------
    list[int]
        Sequence of action indices that (probabilistically) achieve the
        ``make[stick]`` goal.
    """
    # Reset environment to obtain a clean starting state.
    env.reset()

    # Determine the set of possible action indices.
    action_space = list(env.action_specs().values())
    if not action_space:
        return []  # No actions available – return empty plan.

    actions = []
    max_steps = 200  # safety cap to avoid infinite loops.

    for _ in range(max_steps):
        # Sample a random action.
        act = np.random.choice(action_space)
        actions.append(int(act))

        # Execute the action in the environment.
        _, done, _ = env.step(act)

        # Stop once the task reports completion.
        if done:
            break

    # Re‑reset the environment so that the caller can start from the
    # original initial state when replaying the plan.
    env.reset()
    return actions
