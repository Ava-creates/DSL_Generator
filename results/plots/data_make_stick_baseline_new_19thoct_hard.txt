    """
    Generate a list of actions that will collect the required primitives for a
    ``stick`` and then craft it. The function works with the CraftLab
    environment returned by ``env_factory.EnvironmentFactory``.
    """
    # Reset the environment to obtain the initial observation and state.
    obs = env.reset()
    state = env._current_state          # CraftState after reset
    cur_pos = state.pos                  # (x, y) coordinates

    # ------------------------------------------------------------------
    # Helper to generate movement actions between two grid cells.
    # Action mapping (as used by CraftLab):
    #   0: DOWN, 1: UP, 2: LEFT, 3: RIGHT, 4: USE
    # ------------------------------------------------------------------
    def _move_actions(start, target):
        sx, sy = start
        tx, ty = target
        actions = []
        while sx < tx:
            actions.append(3)   # RIGHT
            sx += 1
        while sx > tx:
            actions.append(2)   # LEFT
            sx -= 1
        while sy < ty:
            actions.append(0)   # DOWN
            sy += 1
        while sy > ty:
            actions.append(1)   # UP
            sy -= 1
        return actions

    # ------------------------------------------------------------------
    # Identify indices for the goal item and its required primitives.
    # ------------------------------------------------------------------
    stick_idx = env.world.cookbook.index["stick"]
    # Mapping primitive_kind_index -> required count
    needed_prims = env.world.cookbook.primitives_for(stick_idx)

    # Global egocentric view (one‑hot per kind)
    grid = obs["features_dict"]["features_global"]   # shape (W, H, n_kinds)

    plan = []

    # ------------------------------------------------------------------
    # Collect each needed primitive.
    # ------------------------------------------------------------------
    for prim_idx, count in needed_prims.items():
        for _ in range(count):
            # locate any cell that currently contains this primitive
            locations = np.argwhere(grid[:, :, prim_idx] > 0)
            if locations.size == 0:
                # primitive not present – cannot continue, break early
                continue
            ty, tx = locations[0]          # argwhere returns (row, col)
            plan.extend(_move_actions(cur_pos, (tx, ty)))
            cur_pos = (tx, ty)
            plan.append(4)                  # USE to pick it up
            # mark the cell as empty for subsequent searches
            grid[ty, tx, prim_idx] = 0

    # ------------------------------------------------------------------
    # Find the workshop (if the recipe specifies one) and craft.
    # ------------------------------------------------------------------
    recipe = env.world.cookbook.recipes.get(stick_idx, {})
    workshop_idx = None
    for key, _ in recipe.items():
        if isinstance(key, str):          # workshop name stored as string
            workshop_idx = env.world.cookbook.index[key]
            break

    if workshop_idx is not None:
        # locate a workshop cell
        locations = np.argwhere(grid[:, :, workshop_idx] > 0)
        if locations.size > 0:
            ty, tx = locations[0]
            plan.extend(_move_actions(cur_pos, (tx, ty)))
            cur_pos = (tx, ty)
    # Finally, use the crafting action
    plan.append(4)   # USE to craft the stick

    return plan

    """
    Construct a simple plan that moves the agent to a cell containing a
    ``stick`` primitive (if present), faces the primitive and performs a
    USE action to pick it up.

    The function inspects the underlying ``CraftState`` to locate the
    primitive and then creates a list of movement actions (DOWN=0,
    UP=1, LEFT=2, RIGHT=3) followed by a USE action (4).  It does **not**
    modify the environment – it only returns the action list, which the
    caller will later execute.
    """
    # ------------------------------------------------------------------
    # Helper constants – actions as used by CraftLab
    # ------------------------------------------------------------------
    DOWN, UP, LEFT, RIGHT, USE = 0, 1, 2, 3, 4
    opposite_dir = {DOWN: UP, UP: DOWN, LEFT: RIGHT, RIGHT: LEFT}

    # ------------------------------------------------------------------
    # Grab the current state and world information
    # ------------------------------------------------------------------
    state = env._current_state          # type: CraftState
    world = state.world                 # CraftWorld
    cookbook = world.cookbook

    # Index of the desired item (``stick``)
    try:
        stick_idx = cookbook.index["stick"]
    except Exception:  # pragma: no cover
        # If the name is not present, we cannot craft; return empty plan.
        return []

    # ------------------------------------------------------------------
    # Locate a cell that contains a ``stick`` primitive.
    # ------------------------------------------------------------------
    grid = state.grid                     # (W, H, n_kinds)
    positions = np.argwhere(grid[:, :, stick_idx] > 0)

    # If no stick primitive is present, we cannot craft it – return empty.
    if len(positions) == 0:               # pragma: no cover
        return []

    # Choose the first occurrence (any will do)
    target_x, target_y = positions[0]

    # ------------------------------------------------------------------
    # Determine a free adjacent cell from which we can USE the primitive.
    # ------------------------------------------------------------------
    # Cells that are not traversable (e.g., water, trees) are listed in
    # ``world.non_grabbable_indices``.
    non_grabbable = world.non_grabbable_indices

    def cell_is_free(px, py):
        """Return True if (px,py) is inside the grid and contains no
        non‑grabbable entity."""
        if not (0 <= px < grid.shape[0] and 0 <= py < grid.shape[1]):
            return False
        cell = grid[px, py, :]
        for idx in non_grabbable:
            if cell[idx] > 0:
                return False
        return True

    # Search the four neighbours of the target for a free cell.
    # The tuple also stores the direction the agent must face to look at
    # the target from that neighbour.
    neighbour_info = [
        (target_x - 1, target_y, DOWN),   # agent below target, faces UP (DOWN action moves up)
        (target_x + 1, target_y, UP),     # agent above target, faces DOWN
        (target_x, target_y - 1, RIGHT),  # agent left of target, faces RIGHT
        (target_x, target_y + 1, LEFT),   # agent right of target, faces LEFT
    ]

    adj_x = adj_y = adj_dir = None
    for nx, ny, face_dir in neighbour_info:
        if cell_is_free(nx, ny):
            adj_x, adj_y, adj_dir = nx, ny, face_dir
            break

    # If no free neighbour is found, fall back to the target cell itself
    # (only possible when the target cell itself is free).
    if adj_x is None:
        if cell_is_free(target_x, target_y):
            adj_x, adj_y, adj_dir = target_x, target_y, DOWN
        else:                               # pragma: no cover
            return []

    # ------------------------------------------------------------------
    # Build a Manhattan‑style path from the current position to the
    # chosen adjacent cell.
    # ------------------------------------------------------------------
    cur_x, cur_y = state.pos
    actions = []

    # vertical moves
    while cur_x < adj_x:
        actions.append(DOWN)
        cur_x += 1
    while cur_x > adj_x:
        actions.append(UP)
        cur_x -= 1

    # horizontal moves
    while cur_y < adj_y:
        actions.append(RIGHT)
        cur_y += 1
    while cur_y > adj_y:
        actions.append(LEFT)
        cur_y -= 1

    # ------------------------------------------------------------------
    # Ensure the agent is facing the target before using.
    # The last movement action already set the agent's direction.
    # If that direction is not the required one, we perform a small
    # “turn” by moving one step in the desired direction and immediately
    # stepping back.
    # ------------------------------------------------------------------
    if actions:
        last_dir = actions[-1]
    else:
        # No movement performed yet – we don't know the initial direction.
        # In this case we also execute a turn sequence.
        last_dir = None

    if last_dir != adj_dir:
        # Move one step towards the target direction and step back.
        actions.append(adj_dir)                      # step towards target direction
        actions.append(opposite_dir[adj_dir])        # step back to original cell

    # Finally, interact (USE) to pick up the stick.
    actions.append(USE)

    return actions

    """
    Very simple planner that:
    1. Looks up the primitive(s) required for the goal (stick).
    2. Walks to the nearest instance of each required primitive,
       uses the USE action to pick it up.
    3. If the recipe specifies a workshop, walks to the nearest workshop
       and uses the USE action to craft the item.
    The function returns a list of integer actions (0=DOWN, 1=UP, 2=LEFT,
    3=RIGHT, 4=USE) that can be fed directly to the environment.
    """
    from collections import deque

    # ------------------------------------------------------------------
    # Helper utilities
    # ------------------------------------------------------------------
    # action -> (dx, dy)
    ACTION_DELTAS = {
        0: (0, 1),   # DOWN
        1: (0, -1),  # UP
        2: (-1, 0),  # LEFT
        3: (1, 0),   # RIGHT
    }

    def in_bounds(p, w, h):
        x, y = p
        return 0 <= x < w and 0 <= y < h

    def is_passable(p, grid):
        """A cell is passable if it contains no entities."""
        x, y = p
        return grid[x, y].sum() == 0

    def bfs(start, goal_set, grid):
        """
        Breadth‑first search returning a list of actions that moves the
        agent from ``start`` to any position in ``goal_set`` (set of (x,y)
        tuples).  If no path exists, returns an empty list.
        """
        w, h, _ = grid.shape
        queue = deque()
        queue.append((start, []))
        visited = {start}
        while queue:
            pos, path = queue.popleft()
            if pos in goal_set:
                return path
            for a, (dx, dy) in ACTION_DELTAS.items():
                nxt = (pos[0] + dx, pos[1] + dy)
                if not in_bounds(nxt, w, h):
                    continue
                if nxt in visited:
                    continue
                if not is_passable(nxt, grid):
                    continue
                visited.add(nxt)
                queue.append((nxt, path + [a]))
        return []  # no path found

    # ------------------------------------------------------------------
    # Begin planning
    # ------------------------------------------------------------------
    state = env._current_state          # current CraftState
    world = env.world                    # CraftWorld
    goal = env.task.goal                 # integer index of the goal item

    # 1. Determine required primitive items for the goal
    required = world.cookbook.primitives_for(goal)  # {prim_idx: count}
    # If the goal has no recipe (i.e., it is itself a primitive),
    # ``required`` will be empty – in that case we simply try to pick up
    # the goal item directly.
    if not required:
        required = {goal: 1}

    # 2. Locate all entities of interest on the current grid
    grid = state.grid
    w, h, _ = grid.shape

    # map from kind index -> list of positions where it appears
    kind_positions = {}
    for kind in required.keys():
        pos = list(zip(*np.where(grid[:, :, kind] > 0)))
        kind_positions[kind] = pos

    # also collect workshop positions if the recipe specifies one
    workshop_idx = None
    recipe = world.cookbook.recipes.get(goal, {})
    if isinstance(recipe, dict):
        workshop_idx = recipe.get('_at')
    workshop_positions = []
    if workshop_idx is not None:
        workshop_positions = list(zip(*np.where(grid[:, :, workshop_idx] > 0)))

    actions_plan = []
    cur_pos = state.pos  # (x, y)

    # ------------------------------------------------------------------
    # Helper to move next to a target cell and use it
    # ------------------------------------------------------------------
    def move_and_use(target_pos):
        """Move adjacent to ``target_pos`` (if possible) and append USE."""
        nonlocal cur_pos, actions_plan

        # Determine admissible adjacent cells that are passable
        adjacents = []
        for dx, dy in ACTION_DELTAS.values():
            adj = (target_pos[0] + dx, target_pos[1] + dy)
            if in_bounds(adj, w, h) and is_passable(adj, grid):
                adjacents.append(adj)

        # If there is no free adjacent cell, fall back to the agent's current
        # position (might be already on the same cell – unlikely but safe).
        if not adjacents:
            adjacents = [cur_pos]

        # Choose the adjacent cell with the shortest path from current pos
        best_path = None
        best_adj = None
        for adj in adjacents:
            path = bfs(cur_pos, {adj}, grid)
            if best_path is None or (path and len(path) < len(best_path)):
                best_path = path
                best_adj = adj

        if best_path is None:
            # No path found – give up on this target
            return

        # Execute movement actions
        actions_plan.extend(best_path)
        cur_pos = best_adj

        # Finally use the target
        actions_plan.append(4)   # USE
        # After using, we consider the target collected / crafted.
        # We do NOT update the grid here; we rely on the environment to
        # handle the side‑effects when the actions are actually executed.

    # ------------------------------------------------------------------
    # Collect required primitives
    # ------------------------------------------------------------------
    for prim_idx, needed_cnt in required.items():
        # Find all positions of this primitive on the map
        positions = kind_positions.get(prim_idx, [])
        # Simple greedy: take the nearest still‑available primitive each time
        collected = 0
        used_positions = set()
        while collected < needed_cnt and positions:
            # Choose the nearest position not already used
            best_pos = None
            best_dist = None
            for pos in positions:
                if pos in used_positions:
                    continue
                # Manhattan distance from current position
                d = abs(pos[0] - cur_pos[0]) + abs(pos[1] - cur_pos[1])
                if best_dist is None or d < best_dist:
                    best_dist = d
                    best_pos = pos
            if best_pos is None:
                break  # no more available primitives
            move_and_use(best_pos)
            used_positions.add(best_pos)
            collected += 1

    # ------------------------------------------------------------------
    # Craft at workshop if needed
    # ------------------------------------------------------------------
    if workshop_idx is not None and workshop_positions:
        # pick the nearest workshop
        best_ws = min(workshop_positions,
                      key=lambda p: abs(p[0] - cur_pos[0]) + abs(p[1] - cur_pos[1]))
        move_and_use(best_ws)

    return actions_plan

    ``stick`` recipe and then crafts the stick at the appropriate workshop.
    It works directly on the internal ``CraftLab`` state, building a list of
    actions (DOWN=0, UP=1, LEFT=2, RIGHT=3, USE=4) that can be fed to the

    # Helper constants / functions
    # ----------------------------------------------------------------------
    # Action indices as defined by CraftLab
    DOWN, UP, LEFT, RIGHT, USE = 0, 1, 2, 3, 4
    DIRS = [(0, 1), (0, -1), (-1, 0), (1, 0)]          # (dx, dy) for DOWN,UP,LEFT,RIGHT
    DIR_TO_ACTION = {(0, 1): DOWN, (0, -1): UP, (-1, 0): LEFT, (1, 0): RIGHT}

    from collections import deque
    import numpy as np

    # ----------------------------------------------------------------------
    # Access the current state and world objects
    # ----------------------------------------------------------------------
    state = env._current_state          # CraftState
    world = env.world                    # CraftWorld
    goal_idx = env.task.goal             # index of the desired item (stick)

    # ----------------------------------------------------------------------
    # Determine which primitive items are needed for the goal
    # ----------------------------------------------------------------------
    # ``primitives_for`` returns a dict {primitive_index: count}
    needed_prims = world.cookbook.primitives_for(goal_idx)

    actions = []                         # final action list to return

    # ----------------------------------------------------------------------
    # Simple BFS planner that returns a list of movement actions to reach a
    # target cell (standing on an empty cell).  It avoids any occupied cell.
    # ----------------------------------------------------------------------
    def bfs(start, target_set):
        """Return a list of actions (DOWN/UP/LEFT/RIGHT) that moves from
        ``start`` to any cell in ``target_set``.  Returns None if unreachable."""
        visited = set()
        q = deque()
        q.append((start, []))
        visited.add(start)

        width, height = state.grid.shape[0], state.grid.shape[1]

        while q:
            (x, y), path = q.popleft()
            if (x, y) in target_set:
                return path
            for a, (dx, dy) in enumerate(DIRS):
                nx, ny = x + dx, y + dy
                if 0 <= nx < width and 0 <= ny < height:
                    # cell must be empty (sum over all kinds == 0)
                    if np.sum(state.grid[nx, ny, :]) == 0 and (nx, ny) not in visited:
                        visited.add((nx, ny))
                        q.append(((nx, ny), path + [a]))
        return None

    # ----------------------------------------------------------------------
    # Collect each required primitive (respecting multiplicities)
    # ----------------------------------------------------------------------
    for prim_idx, needed_cnt in needed_prims.items():
        for _ in range(needed_cnt):
            # locate all remaining cells that contain this primitive
            prim_cells = list(zip(*np.where(state.grid[:, :, prim_idx] > 0)))
            if not prim_cells:
                # nothing left – break early
                break

            # Build a set of all empty cells that are adjacent to any primitive cell
            adj_targets = set()
            adj_map = {}                     # maps standing cell -> primitive cell it faces
            for (px, py) in prim_cells:
                for dx, dy in DIRS:
                    nx, ny = px + dx, py + dy
                    if 0 <= nx < state.grid.shape[0] and 0 <= ny < state.grid.shape[1]:
                        if np.sum(state.grid[nx, ny, :]) == 0:
                            adj_targets.add((nx, ny))
                            adj_map[(nx, ny)] = (px, py)

            # Find a path to the nearest reachable adjacent cell
            path = bfs(state.pos, adj_targets)
            if path is None:
                # cannot reach – give up on this primitive
                continue

            # Execute the movement actions (record them and update the simulated state)
            actions.extend(path)
            for a in path:
                dx, dy = DIRS[a]
                state.pos = (state.pos[0] + dx, state.pos[1] + dy)
                state.dir = a

            # We are now standing on a cell that is adjacent to the primitive.
            prim_cell = adj_map[state.pos]

            # Use action to pick up the primitive
            actions.append(USE)

            # Simulate the effect of USE: remove the primitive from the grid and
            # add it to the inventory.
            px, py = prim_cell
            state.grid[px, py, prim_idx] = 0
            state.inventory[prim_idx] += 1

    # ----------------------------------------------------------------------
    # After collecting all primitives, craft the goal at its workshop.
    # ----------------------------------------------------------------------
    # Find which workshop kind the recipe requires (stored under the special key "_at")
    recipe_dict = world.cookbook.recipes.get(goal_idx, {})
    workshop_kind = None
    for k, v in recipe_dict.items():
        if k == "_at":
            workshop_kind = v
            break

    # If the recipe does not require a workshop, we are done.
    if workshop_kind is None:
        return actions

    # Locate all workshop cells of that kind
    workshop_cells = list(zip(*np.where(state.grid[:, :, workshop_kind] > 0)))
    if not workshop_cells:
        # No workshop present – cannot craft, just return what we have.
        return actions

    # Find empty cells adjacent to any workshop cell
    workshop_adj = set()
    workshop_adj_map = {}
    for (wx, wy) in workshop_cells:
        for dx, dy in DIRS:
            nx, ny = wx + dx, wy + dy
            if 0 <= nx < state.grid.shape[0] and 0 <= ny < state.grid.shape[1]:
                if np.sum(state.grid[nx, ny, :]) == 0:
                    workshop_adj.add((nx, ny))
                    workshop_adj_map[(nx, ny)] = (wx, wy)

    # Move to the nearest adjacent cell
    path = bfs(state.pos, workshop_adj)
    if path is not None:
        actions.extend(path)
        for a in path:
            dx, dy = DIRS[a]
            state.pos = (state.pos[0] + dx, state.pos[1] + dy)
            state.dir = a

        # Use action to craft at the workshop
        actions.append(USE)

        # Simulate crafting: consume required primitives and add the goal item.
        for prim_idx, cnt in needed_prims.items():
            state.inventory[prim_idx] -= cnt
        state.inventory[goal_idx] += 1

    # ----------------------------------------------------------------------
    # Return the complete action sequence
    # ----------------------------------------------------------------------
    return actions

    """
    Very simple planner that:
    1. Reads the recipe for a ``stick`` from the world’s cookbook.
    2. Moves the agent to the location of each required primitive,
       uses the ``USE`` action to pick it up.
    3. Moves to the required workshop and uses ``USE`` to craft the stick.
    The function returns a list of integer actions (0‑DOWN, 1‑UP,
    2‑LEFT, 3‑RIGHT, 4‑USE) that can be fed directly to the environment.
    """
    import numpy as np

    # ------------------------------------------------------------------
    # Helper: translate a Manhattan move into actions and update position.
    # ------------------------------------------------------------------
    def move_to(target, cur_pos, actions):
        """Append actions that move from cur_pos to target (inclusive)."""
        tx, ty = target
        # move in x direction
        while cur_pos[0] < tx:
            actions.append(3)          # RIGHT
            cur_pos[0] += 1
        while cur_pos[0] > tx:
            actions.append(2)          # LEFT
            cur_pos[0] -= 1
        # move in y direction
        while cur_pos[1] < ty:
            actions.append(0)          # DOWN
            cur_pos[1] += 1
        while cur_pos[1] > ty:
            actions.append(1)          # UP
            cur_pos[1] -= 1
        return cur_pos

    # ------------------------------------------------------------------
    # Grab the current state and world objects.
    # ------------------------------------------------------------------
    state = env._current_state          # CraftState instance
    world = state.world                  # CraftWorld instance
    cb = world.cookbook                  # Cookbook with recipes etc.

    # ------------------------------------------------------------------
    # Identify the index for the goal item "stick".
    # ------------------------------------------------------------------
    try:
        stick_idx = cb.index["stick"]
    except Exception:
        # If the name is not directly present, fall back to the task goal.
        # The task name is of the form "make[stick]".
        task_goal = env.task.goal  # e.g. ("make", "stick")
        if isinstance(task_goal, tuple) and len(task_goal) == 2:
            stick_idx = cb.index[task_goal[1]]
        else:
            raise RuntimeError("Could not locate stick index in cookbook.")
    # ------------------------------------------------------------------
    # Retrieve the recipe for a stick.
    # ------------------------------------------------------------------
    recipe = cb.recipes.get(stick_idx, {})

    actions = []                         # final action list
    cur_pos = list(state.pos)            # mutable copy of current (x, y)

    # ------------------------------------------------------------------
    # Collect each primitive ingredient required by the recipe.
    # ------------------------------------------------------------------
    for ing_key, count in recipe.items():
        if ing_key == "_at":
            continue                     # workshop, handled later
        # Ingredient key may be an int index or a name string.
        if isinstance(ing_key, int):
            ing_idx = ing_key
        else:
            ing_idx = cb.index[ing_key]

        # Locate all cells that contain this ingredient.
        locations = np.argwhere(state.grid[:, :, ing_idx] > 0)
        if locations.size == 0:
            continue                     # nothing to collect (should not happen)

        # Collect the required amount (usually 1).
        needed = int(count)
        loc_iter = iter(locations.tolist())
        while needed > 0:
            try:
                target = next(loc_iter)
            except StopIteration:
                break                     # no more instances on the map
            # Move to the cell containing the ingredient.
            cur_pos = move_to(target, cur_pos, actions)
            # Use the item (pick‑up / interact).
            actions.append(4)            # USE
            needed -= 1

    # ------------------------------------------------------------------
    # Move to the workshop specified by the recipe and craft the stick.
    # ------------------------------------------------------------------
    workshop_name = recipe.get("_at")
    if workshop_name:
        wk_idx = cb.index[workshop_name]
        workshop_locs = np.argwhere(state.grid[:, :, wk_idx] > 0)
        if workshop_locs.size > 0:
            target = workshop_locs[0].tolist()
            cur_pos = move_to(target, cur_pos, actions)
            actions.append(4)            # USE to craft

    return actions

    #   3: RIGHT (x + 1)
    #   4: USE

        while x < tx:
            actions.append(ACTION_RIGHT)
            x += 1
        while x > tx:
            actions.append(ACTION_LEFT)
            x -= 1

        while y < ty:
            actions.append(ACTION_DOWN)
            y += 1
        while y > ty:
            actions.append(ACTION_UP)
            y -= 1
        return actions

                actions.append(3)  # RIGHT
                cx += 1
            elif cx > tx:
                actions.append(2)  # LEFT
                cx -= 1
            elif cy < ty:
                actions.append(0)  # DOWN
                cy += 1
            elif cy > ty:
                actions.append(1)  # UP
                cy -= 1

    for prim_idx, cnt in needed.items():

    # ----------------------------------------------------------------------

    # --------------------------------------------------------------------- #
    # The world holds a ``Cookbook`` with a name → index mapping.

        stick_idx = env.world.cookbook.index['stick']

    stick_positions = np.argwhere(stick_layer > 0)   # list of (x, y) cells

    if len(stick_positions) == 0:        # No stick present – nothing to do.

    for kind in env.world.non_grabbable_indices:
        obstacle_mask |= (state.grid[:, :, kind] > 0)

    # 5. Breadth‑first search from the agent’s start position to the stick
    # --------------------------------------------------------------------- #
